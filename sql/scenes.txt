--cleanup scenes

ALTER TABLE scenes 
  DROP COLUMN room_id CASCADE,
  ADD COLUMN created_by UUID REFERENCES auth.users(id) DEFAULT auth.uid();

ALTER TABLE scenes ALTER COLUMN created_by SET NOT NULL;

-- alte Policies weg
DROP POLICY IF EXISTS "room members can view scenes" ON scenes;
DROP POLICY IF EXISTS "owners can insert scenes" ON scenes;
DROP POLICY IF EXISTS "owners can update scenes" ON scenes;
DROP POLICY IF EXISTS "owners can delete scenes" ON scenes;

-- SELECT: User sieht Scenes, die er erstellt hat ODER die in einem Room sind, dem er angehört
CREATE POLICY "users can view accessible scenes"
  ON scenes FOR SELECT
  USING (
    created_by = auth.uid()
    OR EXISTS (
      SELECT 1 FROM scene_rooms sr
      JOIN room_members rm ON rm.room_id = sr.room_id
      WHERE sr.scene_id = scenes.id
        AND rm.user_id = auth.uid()
    )
  );

-- INSERT: jeder eingeloggte User darf eine Scene für sich anlegen
CREATE POLICY "users can insert their own scenes"
  ON scenes FOR INSERT
  WITH CHECK (created_by = auth.uid());

-- UPDATE/DELETE: nur der Ersteller
CREATE POLICY "creators can update scenes"
  ON scenes FOR UPDATE
  USING (created_by = auth.uid());

CREATE POLICY "creators can delete scenes"
  ON scenes FOR DELETE
  USING (created_by = auth.uid());
